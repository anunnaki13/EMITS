    ```python
    #!/usr/bin/env python3
    """
    Regenerate API_REFERENCE.md from live /openapi.json (D-04, D-06).

    Idempotent: hand-curated content between `<!-- BEGIN HAND-CURATED: <name> -->`
    and `<!-- END HAND-CURATED: <name> -->` markers is preserved across runs.
    The auto-generated endpoint table is replaced wholesale on each run.

    Fetch order (5s timeout each):
      1. http://localhost:8013/openapi.json
      2. http://103.150.197.225:8013/openapi.json

    Exit codes:
      0  success — file written (and unchanged on second run)
      1  schema fetch failed
      2  prerequisite ADR missing or path error
    """

    import json
    import re
    import sys
    import urllib.request
    import urllib.error
    from pathlib import Path
    from typing import Any

    HERE = Path(__file__).resolve().parent
    REPO = HERE.parent
    OUT = REPO / "API_REFERENCE.md"

    OPENAPI_URLS = [
        "http://localhost:8013/openapi.json",
        "http://103.150.197.225:8013/openapi.json",
    ]

    BEGIN_GENERATED = "<!-- BEGIN GENERATED -->"
    END_GENERATED = "<!-- END GENERATED -->"

    HAND_CURATED_NAMES = ["header", "auth", "pagination", "errors", "examples"]

    SEED_HEADER = """\
    # API Reference

    Live FastAPI surface for the PLTU Tenayan Fuel Management System. Generated
    from `/openapi.json` by `scripts/regenerate_api_reference.py` (re-run after
    any backend route change). Hand-curated narrative sections live between
    `<!-- BEGIN HAND-CURATED: <name> -->` markers and are preserved across
    regeneration.

    Last regenerated: <see git log -1 -- API_REFERENCE.md>.
    """

    SEED_AUTH = """\
    ## Auth Contract

    See `.planning/decisions/ADR-004-jwt-bcrypt-three-role-auth.md` (locked
    2026-05-10) for the architectural decision; this section restates the
    operational contract for API consumers and points at `docs/audit/AUTH_CONTRACT.md`
    for the Phase-2 reconciliation record (D-AUTH-01 422→400 remap;
    D-AUTH-02 403-on-missing-Authorization disposition).

    - **Token type:** JWT (HS256), issued by `POST /api/auth/login`.
    - **Header:** `Authorization: Bearer <JWT>` on every protected endpoint.
    - **Roles:** `admin`, `operator`, `viewer`. Role enforcement is server-side
      via the `require_role(...)` dependency in `backend/server.py:599`.
    - **Login error mapping:** see `## Error Code Map` below.
    - **Public endpoints (no Bearer required):** `/api/auth/login`,
      `/api/auth/register`, `/api/`, `/api/health`. All other endpoints require
      an Authorization header.
    """

    SEED_PAGINATION = """\
    ## Pagination Contract

    See `.planning/decisions/ADR-008-pagination-shape.md` for the locked envelope
    (CONS-pagination-shape).

    Every paginated list endpoint returns:

    ```json
    {
      "items": [ /* records */ ],
      "total": 461,
      "page": 1,
      "page_size": 50,
      "total_pages": 10
    }
    ```

    - Query params: `page` (default 1), `page_size` (default 50).
    - Operational caps: most list endpoints `page_size <= 500`.
      Smart-stock list (`GET /api/smart-stock`) caps at `limit=50000`.
    - Frontend MUST read `response.data.items`; never assume the response is
      a bare JSON array.
    """

    SEED_ERRORS = """\
    ## Error Code Map

    See `.planning/intel/constraints.md` → CONS-auth-header for the locked
    semantic mapping. The `/api/auth/*` surface returns 400 for malformed body
    (Phase-2 D-AUTH-01 remap from FastAPI default 422); non-auth routes keep
    the FastAPI 422 default for validation failures.

    | Code | Used by | Semantic meaning |
    |------|---------|------------------|
    | 200  | all     | success |
    | 400  | `/api/auth/*` | malformed request body (validation failure) |
    | 401  | `/api/auth/login`, `/api/auth/me` | invalid credentials or invalid/expired token |
    | 403  | any protected endpoint | role denied OR missing Authorization header (HTTPBearer default — D-AUTH-02) |
    | 404  | `*/{id}` lookups | record not found |
    | 422  | non-auth `POST/PUT` endpoints | malformed request body (FastAPI default — preserved for backwards compat) |
    | 500  | AI endpoints | internal error or LLM integration failure (e.g. BudgetExceededError) |
    """

    SEED_EXAMPLES = """\
    ## Per-Module Call Examples

    Credentials in the examples below MUST be sourced from environment
    variables. Never inline literal passwords in this file — the credential
    scanner (`scripts/check_credentials.sh`) blocks committing inline literals.

    ```bash
    # Source admin credentials from local memory/test_credentials.md (gitignored)
    cd pltu-tenayan-full-backup
    export TEST_ADMIN_EMAIL="$(awk '/^## Akun Admin$/,/^##/{ if(/Email:/){sub(/^- Email:[[:space:]]*/,"");print;exit} }' memory/test_credentials.md)"
    export TEST_ADMIN_PASSWORD="$(awk '/^## Akun Admin$/,/^##/{ if(/Password:/){sub(/^- Password:[[:space:]]*/,"");print;exit} }' memory/test_credentials.md)"

    # Login (returns access_token + user envelope)
    curl -fsS -X POST http://103.150.197.225:8013/api/auth/login \\
      -H "Content-Type: application/json" \\
      -d "$(python3 -c "import json,os;print(json.dumps({'email':os.environ['TEST_ADMIN_EMAIL'],'password':os.environ['TEST_ADMIN_PASSWORD']}))")"

    # /me — Bearer-token rehydrate
    curl -fsS http://103.150.197.225:8013/api/auth/me \\
      -H "Authorization: Bearer $TOKEN"

    # Vessels list (paginated)
    curl -fsS "http://103.150.197.225:8013/api/vessels?page=1&page_size=50" \\
      -H "Authorization: Bearer $TOKEN" | python3 -m json.tool | head -20

    unset TEST_ADMIN_EMAIL TEST_ADMIN_PASSWORD
    ```
    """

    SEED_HAND = {
        "header": SEED_HEADER,
        "auth": SEED_AUTH,
        "pagination": SEED_PAGINATION,
        "errors": SEED_ERRORS,
        "examples": SEED_EXAMPLES,
    }

    def fetch_openapi() -> dict:
        last_err = None
        for url in OPENAPI_URLS:
            try:
                with urllib.request.urlopen(url, timeout=5) as resp:
                    return json.loads(resp.read())
            except (urllib.error.URLError, urllib.error.HTTPError, TimeoutError) as e:
                last_err = (url, e)
                continue
        raise SystemExit(f"openapi fetch failed (last: {last_err}). Bring up uvicorn — see LOCAL_SETUP.md §VPS Service Recovery.")

    def render_table(spec: dict) -> str:
        rows: list[str] = []
        rows.append("| Method | Path | Auth | Request schema | Responses |")
        rows.append("|--------|------|------|----------------|-----------|")
        paths = spec.get("paths", {}) or {}
        public = {"/api/auth/login", "/api/auth/register", "/api/", "/api/health"}
        for path in sorted(paths.keys()):
            for method, op in sorted(paths[path].items()):
                if method.lower() not in ("get","post","put","patch","delete"):
                    continue
                req_ref = "—"
                rb = op.get("requestBody")
                if rb:
                    schema = (rb.get("content", {}).get("application/json", {}) or {}).get("schema", {})
                    ref = schema.get("$ref")
                    if ref:
                        req_ref = ref.split("/")[-1]
                    elif schema.get("type"):
                        req_ref = schema["type"]
                resp_codes = ",".join(sorted((op.get("responses", {}) or {}).keys()))
                if path in public and op.get("security") in (None, [], [{}]):
                    auth = "public"
                else:
                    auth = "Bearer" if op.get("security") or path not in public else "public"
                rows.append(f"| {method.upper()} | `{path}` | {auth} | {req_ref} | {resp_codes} |")
        return "\n".join(rows)

    def parse_existing(text: str) -> dict[str, str]:
        out = {}
        for name in HAND_CURATED_NAMES:
            m = re.search(
                rf"<!-- BEGIN HAND-CURATED: {name} -->\n(.*?)\n<!-- END HAND-CURATED: {name} -->",
                text, flags=re.DOTALL)
            if m:
                out[name] = m.group(1)
        return out

    def render_doc(generated: str, hand: dict[str, str]) -> str:
        def block(name: str) -> str:
            body = hand.get(name, SEED_HAND[name])
            return f"<!-- BEGIN HAND-CURATED: {name} -->\n{body}\n<!-- END HAND-CURATED: {name} -->"
        sections = [
            block("header"),
            block("auth"),
            block("pagination"),
            block("errors"),
            "## Endpoint Inventory (auto-generated)",
            "",
            BEGIN_GENERATED,
            generated,
            END_GENERATED,
            block("examples"),
        ]
        return "\n\n".join(sections) + "\n"

    def main() -> int:
        # Sanity: ADR cross-links must exist (Plan 03-01 dependency)
        adr_root = REPO.parent / ".planning" / "decisions"
        for adr in ("ADR-004-jwt-bcrypt-three-role-auth.md", "ADR-008-pagination-shape.md"):
            if not (adr_root / adr).is_file():
                print(f"missing prerequisite: {adr_root / adr}", file=sys.stderr)
                return 2
        spec = fetch_openapi()
        hand = parse_existing(OUT.read_text()) if OUT.is_file() else {}
        doc = render_doc(render_table(spec), hand)
        OUT.write_text(doc)
        print(f"wrote {OUT} ({len(doc)} bytes)")
        return 0

    if __name__ == "__main__":
        raise SystemExit(main())
    ```
