    Implementation outline (the executor writes the actual shell):
    ```bash
    #!/usr/bin/env bash
    set -euo pipefail
    cd "$(git rev-parse --show-toplevel)"

    EXCLUDE=( "scripts/check_credentials.sh" "docs/audit/CREDENTIAL_HYGIENE.md" "docs/audit/LOGIN_BUG.md" )

    # Build target list: tracked files minus exclude list
    targets="$(git ls-files | grep -vxF -f <(printf '%s\n' "${EXCLUDE[@]}") || true)"
    [ -z "$targets" ] && { echo "No tracked files to scan."; exit 0; }

    fail=0
    scan() {
        local label="$1" pattern="$2"
        local hits
        hits="$(printf '%s\n' "$targets" | xargs -I{} grep -lE "$pattern" {} 2>/dev/null || true)"
        if [ -n "$hits" ]; then
            echo "FAIL [$label]: matches found in:"
            printf '  %s\n' $hits
            fail=1
        fi
    }

    scan "JWT" 'JWT_PREFIX[A-Za-z0-9._-]{20,}\.[A-Za-z0-9._-]{20,}\.[A-Za-z0-9._-]{20,}'
    scan "Bearer-JWT" '[Bb]earer[[:space:]]+JWT_PREFIX[A-Za-z0-9._-]{20,}'
    scan "MongoDB-URI-with-creds" 'MONGO_URL_WITH_CREDS_PATTERN'
    scan "Admin-password-literal" '<TEST_ADMIN_PASSWORD>'

    if [ "$fail" -eq 1 ]; then
        echo ""
        echo "Remediation: remove the literal, source from env vars (TEST_ADMIN_PASSWORD etc.), and re-stage."
        echo "Run scripts/check_credentials.sh to re-verify."
        exit 1
    fi

    echo "OK: no tracked credential patterns found in $(git ls-files | wc -l) files."
    ```
